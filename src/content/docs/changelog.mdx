import releases from '../../../content/generated/releases.json';
import highlightsData from '../../../content/generated/changelog-highlights.json';
import ReleaseEntry from '../../../src/components/ReleaseEntry.astro';

export const highlightsMap = Object.fromEntries(highlightsData.map(h => [h.tag_name, h.summary]));

export const eras = [
  { id: 'engine', title: 'Engine Era', range: 'v2.29–v2.58', start: 'v2.29.0', description: 'Reliability, extensibility, web UI, declarative workflows' },
  { id: 'maturity', title: 'Maturity Era', range: 'v2.16–v2.28', start: 'v2.16.0', description: 'Token optimisation, parallel workers, headless mode, quality gates' },
  { id: 'platform', title: 'Platform Era', range: 'v2.8–v2.15', start: 'v2.8.0', description: 'Browser tools, LSP, mac tools, Rust native engine, worktree isolation' },
  { id: 'foundation', title: 'Foundation Era', range: 'v2.3–v2.7', start: 'v2.3.4', description: 'Onboarding, search providers, secret management, monorepo' },
  { id: 'seed', title: 'The Seed', range: 'v0.2–v0.3', start: 'v0.2.9', description: 'Planning scaffolding, milestone tracking, worktree management' },
];

export function getEra(tagName) {
  if (tagName >= 'v2.29') return 'engine';
  if (tagName >= 'v2.16') return 'maturity';
  if (tagName >= 'v2.8') return 'platform';
  if (tagName >= 'v2.3') return 'foundation';
  return 'seed';
}

All releases of GSD 2, from newest to oldest. Each entry shows what was added, changed, and fixed. Click any version number to view the release on GitHub. For the narrative behind these releases, see [The Story of GSD](../story-of-gsd/).

<nav class="era-nav">
  {eras.map(era => (
    <a href={`#${era.id}`} class="era-link">
      <span class="era-range">{era.range}</span>
      <span class="era-desc">{era.description}</span>
    </a>
  ))}
</nav>

{(() => {
  let currentEra = null;
  return releases.map((release, index) => {
    const era = getEra(release.tag_name);
    const eraInfo = eras.find(e => e.id === era);
    const showHeader = era !== currentEra;
    currentEra = era;
    return (
      <>
        {showHeader && eraInfo && (
          <h2 id={eraInfo.id} class="era-header">
            {eraInfo.title} <span class="era-header-range">{eraInfo.range}</span>
          </h2>
        )}
        <ReleaseEntry
          tagName={release.tag_name}
          name={release.name}
          publishedAt={release.published_at}
          htmlUrl={release.html_url}
          body={release.body}
          isLatest={index === 0}
          highlight={highlightsMap[release.tag_name]}
        />
      </>
    );
  });
})()}
